package com.igoo.client;

import com.igoo.client.datagen.*;
import com.igoo.worldgen.ModWorldConfiguredFeatures;
import com.igoo.worldgen.ModWorldPlacedFeatures;
import net.fabricmc.fabric.api.datagen.v1.DataGeneratorEntrypoint;
import net.fabricmc.fabric.api.datagen.v1.FabricDataGenerator;
import net.minecraft.core.RegistrySetBuilder;
import net.minecraft.core.registries.Registries;

public class IgooDataGenerator implements DataGeneratorEntrypoint {

    @Override
    public void onInitializeDataGenerator(FabricDataGenerator fabricDataGenerator) {
        FabricDataGenerator.Pack pack = fabricDataGenerator.createPack();
        
        pack.addProvider(ModRecipeProvider::new);
        pack.addProvider(ModBlockTagsProvider::new);
        pack.addProvider(ModBlockLootTableProvider::new);
        pack.addProvider(ModModelProvider::new);
        pack.addProvider(ModItemTagsProvider::new);
        pack.addProvider(ModWorldgenProvider::new);
        pack.addProvider(ModBiomeTagsProvider::new); 
        
        pack.addProvider(ModChestLootTableProvider::new);
        pack.addProvider(ModEntityLootTableProvider::new);
    }
    
    @Override
    public void buildRegistry(RegistrySetBuilder registryBuilder) {
        // Chỉ đăng ký feature của Mod, bỏ hoàn toàn Vanilla
        registryBuilder.add(Registries.CONFIGURED_FEATURE, ModWorldConfiguredFeatures::configure);
        registryBuilder.add(Registries.PLACED_FEATURE, ModWorldPlacedFeatures::configure);
        registryBuilder.add(Registries.BIOME, ModWorldgenProvider::bootstrapBiomes);
    }
}.

package com.igoo.client.datagen;

import java.util.concurrent.CompletableFuture;

import com.igoo.block.ModBlocks;
import net.fabricmc.fabric.api.datagen.v1.FabricPackOutput;
import net.fabricmc.fabric.api.datagen.v1.provider.FabricBlockLootSubProvider;
import net.minecraft.core.HolderLookup;
import net.minecraft.core.registries.Registries;
import net.minecraft.world.item.Item;
import net.minecraft.world.item.enchantment.Enchantments;
import net.minecraft.world.level.block.Block;
import net.minecraft.world.level.storage.loot.LootTable;
import net.minecraft.world.level.storage.loot.entries.LootItem;
import net.minecraft.world.level.storage.loot.functions.ApplyBonusCount;
import net.minecraft.world.level.storage.loot.functions.SetItemCountFunction;
import net.minecraft.world.level.storage.loot.providers.number.UniformGenerator;

public class ModBlockLootTableProvider extends FabricBlockLootSubProvider {

    public ModBlockLootTableProvider(
            FabricPackOutput packOutput,
            CompletableFuture<HolderLookup.Provider> registriesFuture) {
        super(packOutput, registriesFuture);
    }

    
    
    @Override
    public void generate() {
        ModBlocks.ALL_METALS.forEach(this::dropSelf);
        ModBlocks.ALL_ORES.forEach(ore -> {
            var drop = ModBlocks.ORE_DROPS.get(ore);
            if (drop != null) {
                add(ore, createMultipleOreDrops(ore, drop.raw(), drop.min(), drop.max()));
            }
        });
    }

    public LootTable.Builder createMultipleOreDrops(Block block, Item item, float minDrops, float maxDrops) {
        var enchantments = this.registries.lookupOrThrow(Registries.ENCHANTMENT);

        return this.createSilkTouchDispatchTable(
                block,
                this.applyExplosionDecay(
                        block,
                        LootItem.lootTableItem(item)
                                .apply(SetItemCountFunction.setCount(
                                        UniformGenerator.between(minDrops, maxDrops)))
                                .apply(ApplyBonusCount.addOreBonusCount(
                                        enchantments.getOrThrow(Enchantments.FORTUNE))
                                )
                )
        );
    }
}.

package com.igoo.client.datagen;

import java.util.concurrent.CompletableFuture;

import com.igoo.block.ModBlocks;

import net.fabricmc.fabric.api.datagen.v1.FabricPackOutput;
import net.fabricmc.fabric.api.datagen.v1.provider.FabricTagsProvider;
import net.minecraft.core.HolderLookup;
import net.minecraft.tags.BlockTags;
import net.minecraft.core.registries.BuiltInRegistries;

public class ModBlockTagsProvider extends FabricTagsProvider.BlockTagsProvider {

    public ModBlockTagsProvider(FabricPackOutput output, CompletableFuture<HolderLookup.Provider> registryLookupFuture) {
        super(output, registryLookupFuture);
    }

    
    @Override
    protected void addTags(HolderLookup.Provider registries) {
        var pickaxe = builder(BlockTags.MINEABLE_WITH_PICKAXE);
        var stoneTool = builder(BlockTags.NEEDS_STONE_TOOL);
        var ironTool = builder(BlockTags.NEEDS_IRON_TOOL);
        var diamondTool = builder(BlockTags.NEEDS_DIAMOND_TOOL);

        ModBlocks.ALL_BLOCKS.forEach(block -> {
            var key = BuiltInRegistries.BLOCK.getResourceKey(block).orElseThrow();

            // Tất cả block đều đào bằng cuốc
            pickaxe.add(key);

            // Phân loại cấp công cụ
            if (ModBlocks.MINE_WITH_STONE.contains(block)) {
                stoneTool.add(key);
            } else if (ModBlocks.MINE_WITH_IRON.contains(block)) {
                ironTool.add(key);
            } else if (ModBlocks.MINE_WITH_DIAMOND.contains(block)) {
                diamondTool.add(key);
            }
        });
    }
}.

public class ModChestLootTableProvider extends SimpleFabricLootTableSubProvider {

    public ModChestLootTableProvider(FabricPackOutput output, CompletableFuture<HolderLookup.Provider> registryLookup) {
        super(output, registryLookup, LootContextParamSets.CHEST);
    }

    @Override
    public void generate(BiConsumer<ResourceKey<LootTable>, LootTable.Builder> consumer) {
        
        // ==========================================
        // 1. RƯƠNG TRASH HOME (Cũ)
        // ==========================================
        consumer.accept(ModLootTables.TRASH_HOME_CHEST, LootTable.lootTable()
                // 10% ra 1-5 iron_ingot
                .withPool(LootPool.lootPool()
                        .setRolls(ConstantValue.exactly(1.0F))
                        .when(LootItemRandomChanceCondition.randomChance(0.80F))
                        .add(LootItem.lootTableItem(Items.IRON_INGOT)
                                .apply(SetItemCountFunction.setCount(UniformGenerator.between(1.0F, 5.0F))))
                )
                // 20% ra 1-3 ingot mod
                .withPool(LootPool.lootPool()
                        .setRolls(ConstantValue.exactly(1.0F))
                        .when(LootItemRandomChanceCondition.randomChance(1.0F))
                        .add(LootItem.lootTableItem(ModItems.Aluminum_ingot).apply(SetItemCountFunction.setCount(UniformGenerator.between(1.0F, 3.0F))))
                        .add(LootItem.lootTableItem(ModItems.Tin_ingot).apply(SetItemCountFunction.setCount(UniformGenerator.between(1.0F, 3.0F))))
                        .add(LootItem.lootTableItem(ModItems.Titanium_ingot).apply(SetItemCountFunction.setCount(UniformGenerator.between(1.0F, 2.0F))))
                        .add(LootItem.lootTableItem(ModItems.Tungsten_ingot).apply(SetItemCountFunction.setCount(UniformGenerator.between(0.0F, 1.0F))))
                )
                // 100% ra 1-5 bánh mì pháp
                .withPool(LootPool.lootPool()
                        .setRolls(ConstantValue.exactly(1.0F))
                        .add(LootItem.lootTableItem(ModItems.Baguette)
                                .apply(SetItemCountFunction.setCount(UniformGenerator.between(1.0F, 5.0F))))
                )
                // 1% ra 1 titanium_alloy_block
                .withPool(LootPool.lootPool()
                        .setRolls(ConstantValue.exactly(1.0F))
                        .when(LootItemRandomChanceCondition.randomChance(0.01F))
                        .add(LootItem.lootTableItem(ModBlocks.Titanium_alloy_block))
                )
        );


        // ==========================================
        // 2. RƯƠNG HEROBRINE ALTAR (Mới)
        // ==========================================
        consumer.accept(ModLootTables.HEROBRINE_ALTAR_CHEST, LootTable.lootTable()

    public ModItemTagsProvider(FabricPackOutput output, CompletableFuture<HolderLookup.Provider> registriesFuture) {
        super(output, registriesFuture);
    }

    @Override
    protected void addTags(HolderLookup.Provider registries) {
        // 1. Gắn công cụ vào tag tương ứng của Vanilla

            key(ModItems.Aluminum_sword), key(ModItems.Tin_sword), key(ModItems.Chromium_sword), key(ModItems.Titanium_sword), key(ModItems.Tungsten_sword)
        );
        builder(ItemTags.PICKAXES).add(
            key(ModItems.Aluminum_pickaxe), key(ModItems.Tin_pickaxe), key(ModItems.Chromium_pickaxe), key(ModItems.Titanium_pickaxe), key(ModItems.Tungsten_pickaxe)
        );
        builder(ItemTags.AXES).add(
            key(ModItems.Aluminum_axe), key(ModItems.Tin_axe), key(ModItems.Chromium_axe), key(ModItems.Titanium_axe), key(ModItems.Tungsten_axe)
        );
        builder(ItemTags.SHOVELS).add(
            key(ModItems.Aluminum_shovel), key(ModItems.Tin_shovel), key(ModItems.Chromium_shovel), key(ModItems.Titanium_shovel), key(ModItems.Tungsten_shovel)
        );
        builder(ItemTags.HOES).add(
            key(ModItems.Aluminum_hoe), key(ModItems.Tin_hoe), key(ModItems.Chromium_hoe), key(ModItems.Titanium_hoe), key(ModItems.Tungsten_hoe)
        );

        // 2. Đăng ký tag sửa chữa (Repair) cho từng kim loại
        addRepairTag("aluminum_ingot", ModItems.Aluminum_ingot);
        addRepairTag("tin_ingot", ModItems.Tin_ingot);
        addRepairTag("chromium_ingot", ModItems.Chromium_ingot);
        addRepairTag("titanium_ingot", ModItems.Titanium_ingot);
        addRepairTag("tungsten_ingot", ModItems.Tungsten_ingot);
    }

    private void addRepairTag(String tagName, Item item) {
        TagKey<Item> tag = TagKey.create(Registries.ITEM, Identifier.fromNamespaceAndPath(Igoo.MOD_ID, tagName));
        builder(tag).add(key(item));
    }

    /** Helper: Chuyển đổi Item thành ResourceKey<Item> cho API 26.2 */
    private static ResourceKey<Item> key(Item item) {
        return BuiltInRegistries.ITEM.getResourceKey(item).orElseThrow();
    }
}.

package com.igoo.client.datagen;

import com.igoo.block.ModBlocks;
import com.igoo.item.ModItems;
import net.fabricmc.fabric.api.client.datagen.v1.provider.FabricModelProvider;
import net.fabricmc.fabric.api.datagen.v1.FabricPackOutput;
import net.minecraft.client.data.models.BlockModelGenerators;
import net.minecraft.client.data.models.ItemModelGenerators;
import net.minecraft.client.data.models.model.ModelTemplates;

public class ModModelProvider extends FabricModelProvider {
    public ModModelProvider(FabricPackOutput output) {
        super(output);
    }

    @Override
    public void generateBlockStateModels(BlockModelGenerators blockModelGenerator) {
        ModBlocks.ALL_BLOCKS.forEach(block -> blockModelGenerator.createTrivialCube(block));
    }

    @Override
    public void generateItemModels(ItemModelGenerators itemModelGenerator) {
        ModItems.ALL_ITEMS.forEach(item -> itemModelGenerator.generateFlatItem(item, ModelTemplates.FLAT_ITEM));

import net.fabricmc.fabric.api.datagen.v1.FabricPackOutput;

import net.minecraft.world.item.Items;
// import net.minecraft.world.item.crafting.BlastingRecipe;
import net.minecraft.world.item.crafting.CampfireCookingRecipe;
import net.minecraft.world.item.crafting.CookingBookCategory;
import net.minecraft.world.item.crafting.SmeltingRecipe;
import net.minecraft.world.item.crafting.SmokingRecipe;
import net.minecraft.world.level.ItemLike;
import net.minecraft.world.level.block.Block;

public class ModRecipeProvider extends FabricRecipeProvider {

    public ModRecipeProvider(FabricPackOutput output,
                             CompletableFuture<HolderLookup.Provider> registriesFuture) {
        super(output, registriesFuture);
    }

    @Override
    protected RecipeProvider createRecipeProvider(HolderLookup.Provider registries, RecipeOutput output) {
        return new RecipeProvider(registries, output) {
            
            // 👇 KHAI BÁO HÀM COOKFOOD Ở ĐÂY (Ngoài buildRecipes)
            private void cookFood(List<ItemLike> inputs, ItemLike result, String groupName) {
                // 1. Smelting
                oreCooking(SmeltingRecipe::new, inputs, RecipeCategory.FOOD, CookingBookCategory.FOOD, 
                           result, 0.35f, 200, groupName, "_from_smelting");
                // 2. Smoking
                oreCooking(SmokingRecipe::new, inputs, RecipeCategory.FOOD, CookingBookCategory.FOOD, 
                           result, 0.35f, 100, groupName, "_from_smoking");
                // 3. Campfire
                oreCooking(CampfireCookingRecipe::new, inputs, RecipeCategory.FOOD, CookingBookCategory.FOOD, 
                           result, 0.35f, 600, groupName, "_from_campfire_cooking");
            }

            @Override
            public void buildRecipes() {
                // ===== Nấu chảy: raw + ore -> ingot =====
                for (Block ore : ModBlocks.ALL_ORES) {
                    String base = stripSuffix(path(ore), "_ore");
                    Item raw   = findItem("raw_" + base);
                    Item ingot = findItem(base + "_ingot");
                    if (raw == null || ingot == null) continue;

                    List<ItemLike> smeltables = List.of(raw, ore);

                    oreSmelting(smeltables, RecipeCategory.MISC, CookingBookCategory.MISC, ingot, 0.25f, 200, base);
                    oreBlasting(smeltables, RecipeCategory.MISC, CookingBookCategory.MISC, ingot, 0.25f, 100, base);
                }

                // ===== Nén: 9 ingot <-> 1 block =====
                for (Block block : ModBlocks.ALL_METALS) {
                    String base = stripSuffix(path(block), "_block");
                    Item ingot = findItem(base + "_ingot");
                    if (ingot == null) ingot = findItem(base);
                    if (ingot == null) continue;

                    nineBlockStorageRecipes(RecipeCategory.MISC, ingot, RecipeCategory.MISC, block);
                }
                
                // ===== Dụng cụ =====
                record ToolSet(Item ingot, Item sword, Item pickaxe, Item axe, Item shovel, Item hoe) {}

                List<ToolSet> toolSets = List.of(
                    new ToolSet(ModItems.Aluminum_ingot, ModItems.Aluminum_sword, ModItems.Aluminum_pickaxe, ModItems.Aluminum_axe, ModItems.Aluminum_shovel, ModItems.Aluminum_hoe),
                    new ToolSet(ModItems.Tin_ingot, ModItems.Tin_sword, ModItems.Tin_pickaxe, ModItems.Tin_axe, ModItems.Tin_shovel, ModItems.Tin_hoe),
                    new ToolSet(ModItems.Chromium_ingot, ModItems.Chromium_sword, ModItems.Chromium_pickaxe, ModItems.Chromium_axe, ModItems.Chromium_shovel, ModItems.Chromium_hoe),
                    new ToolSet(ModItems.Titanium_ingot, ModItems.Titanium_sword, ModItems.Titanium_pickaxe, ModItems.Titanium_axe, ModItems.Titanium_shovel, ModItems.Titanium_hoe),
                    new ToolSet(ModItems.Tungsten_ingot, ModItems.Tungsten_sword, ModItems.Tungsten_pickaxe, ModItems.Tungsten_axe, ModItems.Tungsten_shovel, ModItems.Tungsten_hoe)
                );

                for (ToolSet set : toolSets) {
                    String axeName = BuiltInRegistries.ITEM.getKey(set.axe()).getPath();
                    String hoeName = BuiltInRegistries.ITEM.getKey(set.hoe()).getPath();

                    // Sword
                    shaped(RecipeCategory.COMBAT, set.sword())
                        .pattern("X").pattern("X").pattern("|")
                        .define('X', set.ingot()).define('|', Items.STICK)
                        .unlockedBy(getHasName(set.ingot()), has(set.ingot())).save(output);

                    // Pickaxe
                    shaped(RecipeCategory.TOOLS, set.pickaxe())
                        .pattern("XXX").pattern(" | ").pattern(" | ")
                        .define('X', set.ingot()).define('|', Items.STICK)
                        .unlockedBy(getHasName(set.ingot()), has(set.ingot())).save(output);

                    // Axe (2 kiểu)
                    shaped(RecipeCategory.TOOLS, set.axe())
                        .pattern("XX").pattern("X|").pattern(" |")
                        .define('X', set.ingot()).define('|', Items.STICK)
                        .unlockedBy(getHasName(set.ingot()), has(set.ingot())).save(output, axeName);

                    shaped(RecipeCategory.TOOLS, set.axe())
                        .pattern("XX").pattern("|X").pattern("| ")
                        .define('X', set.ingot()).define('|', Items.STICK)
                        .unlockedBy(getHasName(set.ingot()), has(set.ingot())).save(output, axeName + "_mirrored");

                    // Shovel
                    shaped(RecipeCategory.TOOLS, set.shovel())
                        .pattern("X").pattern("|").pattern("|")
                        .define('X', set.ingot()).define('|', Items.STICK)
                        .unlockedBy(getHasName(set.ingot()), has(set.ingot())).save(output);

                    // Hoe (2 kiểu)
                    shaped(RecipeCategory.TOOLS, set.hoe())
                        .pattern("XX").pattern(" |").pattern(" |")
                        .define('X', set.ingot()).define('|', Items.STICK)
                        .unlockedBy(getHasName(set.ingot()), has(set.ingot())).save(output, hoeName);

                    shaped(RecipeCategory.TOOLS, set.hoe())
                        .pattern("XX").pattern("| ").pattern("| ")
                        .define('X', set.ingot()).define('|', Items.STICK)
                        .unlockedBy(getHasName(set.ingot()), has(set.ingot())).save(output, hoeName + "_mirrored");
                }

                // ===== THỨC ĂN =====
                List<ItemLike> chickenSmeltables = List.of(ModItems.Chicken_wing);
                List<ItemLike> porkSmeltables = List.of(ModItems.Pork);
                List<ItemLike> baguetteSmeltables = List.of(Items.HAY_BLOCK);

                cookFood(chickenSmeltables, ModItems.Chicken_wing_cooked, "chicken_wing");
                cookFood(porkSmeltables, ModItems.Pork_cooked, "pork");
                cookFood(baguetteSmeltables, ModItems.Baguette, "baguette");

                // ===== PHÔ MAI =====
                shapeless(RecipeCategory.FOOD, ModItems.Cheese_full)
                    .requires(Items.MILK_BUCKET)
                    .requires(Items.WHEAT, 3)
                    .unlockedBy("has_milk", has(Items.MILK_BUCKET))
                    .save(output);

                shaped(RecipeCategory.FOOD, ModItems.Cheese_full)
                    .pattern("CCC").pattern("CCC").pattern("CCC")
                    .define('C', ModItems.Cheese_slice)
                    .unlockedBy(getHasName(ModItems.Cheese_slice), has(ModItems.Cheese_slice))
                    .save(output, "cheese_full_from_slices");

                shapeless(RecipeCategory.FOOD, ModItems.Cheese_slice, 9)
                    .requires(ModItems.Cheese_full)
                    .unlockedBy(getHasName(ModItems.Cheese_full), has(ModItems.Cheese_full))
                    .save(output);
            }
        };
    }

    @Override
    public String getName() {
        return "IGOO Recipes";
    }

    // --- Các hàm helper ---
    private static String path(Block block) {
        return BuiltInRegistries.BLOCK.getKey(block).getPath();
    }

    private static String stripSuffix(String name, String suffix) {
        return name.endsWith(suffix) ? name.substring(0, name.length() - suffix.length()) : name;
    }

    private static Item findItem(String id) {
        return BuiltInRegistries.ITEM
                .getOptional(Identifier.fromNamespaceAndPath(Igoo.MOD_ID, id))
                .orElse(null);
    }
}.

package com.igoo.client.datagen;

import com.igoo.worldgen.ModBiomes;
import net.fabricmc.fabric.api.datagen.v1.FabricPackOutput;
import net.fabricmc.fabric.api.datagen.v1.provider.FabricDynamicRegistryProvider;
import net.minecraft.core.HolderLookup;
import net.minecraft.core.registries.Registries;
import net.minecraft.data.worldgen.BootstrapContext;
import net.minecraft.world.attribute.EnvironmentAttributes;
import net.minecraft.world.entity.EntityTypes;
import net.minecraft.world.entity.MobCategory;
import net.minecraft.world.level.biome.*;

import java.util.concurrent.CompletableFuture;

public class ModWorldgenProvider extends FabricDynamicRegistryProvider {

    public ModWorldgenProvider(FabricPackOutput output, CompletableFuture<HolderLookup.Provider> registriesFuture) {
        super(output, registriesFuture);
    }

    @Override
    protected void configure(HolderLookup.Provider registries, Entries entries) {
        entries.addAll(registries.lookupOrThrow(Registries.CONFIGURED_FEATURE));
        entries.addAll(registries.lookupOrThrow(Registries.PLACED_FEATURE));
        entries.addAll(registries.lookupOrThrow(Registries.BIOME));
    }

    public static void bootstrapBiomes(BootstrapContext<Biome> context) {
        var placedFeatureGetter = context.lookup(Registries.PLACED_FEATURE);
        var carverGetter = context.lookup(Registries.CONFIGURED_CARVER);

        // 1. Hiệu ứng màu sắc
        BiomeSpecialEffects.Builder effects = new BiomeSpecialEffects.Builder()
                .waterColor(0x8B0000)
                .grassColorOverride(0x550000)
                .foliageColorOverride(0x550000);

        // 3. Cấu hình sinh thế giới (Rỗng, vì feature đã được thêm qua BiomeModifications)
        BiomeGenerationSettings generation = new BiomeGenerationSettings.Builder(placedFeatureGetter, carverGetter)
                .build();

        // 4. Cấu hình sinh vật
        MobSpawnSettings spawns = new MobSpawnSettings.Builder()
                .addSpawn(MobCategory.MONSTER, 100, new MobSpawnSettings.SpawnerData(EntityTypes.ZOMBIE, 4, 4))
                .addSpawn(MobCategory.MONSTER, 100, new MobSpawnSettings.SpawnerData(EntityTypes.SKELETON, 4, 4))
                .addSpawn(MobCategory.MONSTER, 20, new MobSpawnSettings.SpawnerData(EntityTypes.WITCH, 1, 2))
                .addSpawn(MobCategory.CREATURE, 5, new MobSpawnSettings.SpawnerData(EntityTypes.HORSE, 2, 6))
                .build();

        // 5. Đăng ký Biome
        context.register(ModBiomes.CURSED_SAVANNA, new Biome.BiomeBuilder()
                .hasPrecipitation(false)
                .temperature(2.0f)
                .downfall(0.0f)
                .specialEffects(effects.build())
                .setAttribute(EnvironmentAttributes.SKY_COLOR, Integer.valueOf(0xFF8B0000))
                .setAttribute(EnvironmentAttributes.FOG_COLOR, Integer.valueOf(0xFF8B0000))
                .setAttribute(EnvironmentAttributes.WATER_FOG_COLOR, Integer.valueOf(0xFF8B0000))
                .mobSpawnSettings(spawns)
                .generationSettings(generation)
                .build());
    }

    @Override
    public String getName() {
        return "Igoo Worldgen";
    }
}.

package com.igoo.command;

import com.mojang.brigadier.arguments.IntegerArgumentType;
import com.mojang.brigadier.builder.LiteralArgumentBuilder;
import com.mojang.brigadier.context.CommandContext;
import net.minecraft.commands.CommandBuildContext;
import net.minecraft.commands.CommandSourceStack;
import net.minecraft.commands.Commands;
import net.minecraft.commands.arguments.ResourceArgument;
import net.minecraft.core.Holder;
import net.minecraft.core.registries.Registries;
import net.minecraft.network.chat.Component;
import net.minecraft.server.level.ServerPlayer;
import net.minecraft.world.item.ItemStack;
import net.minecraft.world.item.enchantment.Enchantment;
import net.minecraft.world.item.enchantment.EnchantmentHelper;

public class EnchantCommand {

    private static final String NAME = "enchant";
    private static final int MIN_LEVEL = 1;
    private static final int MAX_LEVEL = 255;

    /**
     * Build subcommand tree:
     * <pre>
     *   /igoo enchant &lt;enchantment&gt; &lt;level&gt;
     * </pre>
     */
    public static LiteralArgumentBuilder<CommandSourceStack> build(CommandBuildContext ctx) {
        return Commands.literal(NAME)
                .then(Commands.argument("enchantment",
                                ResourceArgument.resource(ctx, Registries.ENCHANTMENT))
                        .then(Commands.argument("level",
                                        IntegerArgumentType.integer(MIN_LEVEL, MAX_LEVEL))
                                .executes(EnchantCommand::execute)));
    }

    private static int execute(CommandContext<CommandSourceStack> ctx) {
        CommandSourceStack source = ctx.getSource();

        // Phải là player (không dùng từ console)
        if (!(source.getEntity() instanceof ServerPlayer player)) {
            source.sendFailure(Component.literal("§cLệnh này chỉ dùng được bởi người chơi!"));
            return 0;
        }

        // Tay chính phải cầm item
        ItemStack stack = player.getMainHandItem();
        if (stack.isEmpty()) {
            source.sendFailure(Component.literal("§cTay chính phải cầm vật phẩm!"));
            return 0;
        }

        // Lấy enchantment
        Holder<Enchantment> enchantment;
        try {
            enchantment = ResourceArgument.getEnchantment(ctx, "enchantment");
        } catch (Exception e) {
            source.sendFailure(Component.literal("§cKhông tìm thấy enchantment!"));
            return 0;
        }

        int level = IntegerArgumentType.getInteger(ctx, "level");

        // Enchant bất kỳ item
        EnchantmentHelper.updateEnchantments(stack,
                mutable -> mutable.set(enchantment, level));

        // Thông báo
        String enchantName = enchantment.value().description().getString();
        String itemName = stack.getHoverName().getString();
        source.sendSuccess(() -> Component.literal(
                "§aĐã enchant §e" + enchantName + " " + toRoman(level)
                        + " §avào §e" + itemName + " §a(level " + level + ")"
        ), false);

        return 1;
    }

    /** Chuyển số → số La Mã (1–255) */
    private static String toRoman(int n) {
        if (n <= 0 || n > 3999) return String.valueOf(n);

        int[] values =    {1000, 900, 500, 400, 100, 90, 50, 40, 10, 9, 5, 4, 1};
        String[] symbols = {"M",  "CM", "D", "CD", "C", "XC", "L", "XL", "X", "IX", "V", "IV", "I"};

        StringBuilder sb = new StringBuilder();
        for (int i = 0; i < values.length; i++) {
            while (n >= values[i]) {
                sb.append(symbols[i]);
                n -= values[i];
            }
        }
        return sb.toString();
    }
}.

    private static int executeList(CommandContext<CommandSourceStack> ctx) {
        List<BotPlayer> bots = collectBots(ctx);
        CommandSourceStack source = ctx.getSource();

        if (bots.isEmpty()) {
            source.sendSuccess(() -> Component.literal("§eKhông có bot nào."), false);
            return 0;
        }

        source.sendSuccess(() -> Component.literal(
                "§6=== Bot đang active (" + bots.size() + ") ==="), false);

        for (BotPlayer bot : bots) {
            BlockPos pos = bot.blockPosition();
            // 👈 getName() → name()
            String botName = bot.getGameProfile().name();
            source.sendSuccess(() -> Component.literal(String.format(
                    "§e- §f%s §7| §amode=§f%s §7| §f(%d, %d, %d) §7| §c%.0f/%.0f HP",
                    botName,
                    bot.getMode().name().toLowerCase(),
                    pos.getX(), pos.getY(), pos.getZ(),
                    bot.getHealth(), bot.getMaxHealth())), false);
        }
        return bots.size();
    }

    private static int executeRemove(CommandContext<CommandSourceStack> ctx) {
        String name = StringArgumentType.getString(ctx, "name");
        List<BotPlayer> toRemove = new ArrayList<>();
        CommandSourceStack source = ctx.getSource();

        if (name.equalsIgnoreCase("all")) {
            toRemove.addAll(collectBots(ctx));
        } else {
            ServerPlayer p = source.getServer().getPlayerList().getPlayerByName(name);
            if (p instanceof BotPlayer bot) toRemove.add(bot);
        }

        if (toRemove.isEmpty()) {
            source.sendFailure(Component.literal("§cKhông tìm thấy bot: " + name));
            return 0;
        }

        for (BotPlayer bot : toRemove) {
            source.getServer().getPlayerList().remove(bot);
        }
        final int count = toRemove.size();
        source.sendSuccess(() -> Component.literal("§aĐã xóa §e" + count + " §abot."), false);
        return count;
    }

    private static int setModeFor(CommandContext<CommandSourceStack> ctx, String botName) {
        String modeName = StringArgumentType.getString(ctx, "mode").toUpperCase();
        BotPlayer.Mode mode;
        try {
            mode = BotPlayer.Mode.valueOf(modeName);
        } catch (IllegalArgumentException e) {
            ctx.getSource().sendFailure(Component.literal("§cMode không hợp lệ: " + modeName));
            return 0;
        }

        int changed = 0;
        for (BotPlayer bot : collectBots(ctx)) {
            // 👈 getName() → name()
            if (botName == null || botName.equalsIgnoreCase("all")
                    || bot.getGameProfile().name().equalsIgnoreCase(botName)) {
                bot.setMode(mode);
                changed++;
            }
        }

        if (changed == 0) {
            ctx.getSource().sendFailure(Component.literal("§cKhông tìm thấy bot: " + botName));
            return 0;
        }
        final int c = changed;
        final String target = botName == null ? "tất cả" : botName;
        ctx.getSource().sendSuccess(() -> Component.literal(
                "§aĐổi mode §e" + mode.name().toLowerCase() + " §acho §e" + target
                        + " §a(" + c + " bot)"), false);
        return changed;
    }

    private static List<BotPlayer> collectBots(CommandContext<CommandSourceStack> ctx) {
        List<BotPlayer> bots = new ArrayList<>();
        for (ServerPlayer p : ctx.getSource().getServer().getPlayerList().getPlayers()) {
            if (p instanceof BotPlayer bot) bots.add(bot);
        }
        return bots;
    }

    private static CompletableFuture<Suggestions> suggestModes(
            CommandContext<CommandSourceStack> ctx, SuggestionsBuilder builder) {
        String rem = builder.getRemaining().toLowerCase();
        for (BotPlayer.Mode m : BotPlayer.Mode.values()) {
            String n = m.name().toLowerCase();
            if (n.startsWith(rem)) builder.suggest(n);
        }
        return builder.buildFuture();
    }

    private static CompletableFuture<Suggestions> suggestBotNames(
            CommandContext<CommandSourceStack> ctx, SuggestionsBuilder builder) {
        String rem = builder.getRemaining().toLowerCase();
        if ("all".startsWith(rem)) builder.suggest("all");
        for (ServerPlayer p : ctx.getSource().getServer().getPlayerList().getPlayers()) {
            if (p instanceof BotPlayer) {
                // 👈 getName() → name()
                String n = p.getGameProfile().name();
                if (n.toLowerCase().startsWith(rem)) builder.suggest(n);
            }
        }
        return builder.buildFuture();
    }

public class FakePlayerManager {

    private static final Random RANDOM = new Random();
    private static final String NAME_PREFIX = "Bot_";

    public static int spawnFakePlayers(MinecraftServer server, ServerPlayer spawner, int count) {
        if (!server.isPublished()) return -1;

        PlayerList playerList = server.getPlayerList();
        ServerLevel level = server.overworld();
        int spawned = 0;

        for (int i = 0; i < count; i++) {
            String name = generateUniqueName(playerList);
            UUID uuid = UUID.nameUUIDFromBytes(name.getBytes());
            GameProfile profile = new GameProfile(uuid, name);

            BotPlayer bot = new BotPlayer(level, profile);
            bot.setOwner(spawner);   // 👈 bot biết ai là chủ

            // Give vật phẩm để bot có thể xây / đánh
            bot.getInventory().add(new ItemStack(Items.STONE, 64));
            bot.getInventory().add(new ItemStack(Items.DIRT, 64));
            bot.getInventory().add(new ItemStack(Items.COBBLESTONE, 64));
            bot.getInventory().add(new ItemStack(Items.DIAMOND_SWORD));
            bot.getInventory().add(new ItemStack(Items.IRON_PICKAXE));

            FakeClientConnection connection = new FakeClientConnection();
            try {
                CommonListenerCookie cookie = CommonListenerCookie.createInitial(profile, false);
                playerList.placeNewPlayer(connection, bot, cookie);
                spawned++;
            } catch (Exception e) {
                e.printStackTrace();
            }
        }
        return spawned;
    }

    private static String generateUniqueName(PlayerList playerList) {
        String name;
        do { name = NAME_PREFIX + RANDOM.nextInt(10000); }
        while (playerList.getPlayerByName(name) != null);
        return name;
    }

     */
    public boolean tryEquipArmor(ItemStack stack) {
        if (stack.isEmpty()) return false;

        // Xác định slot từ EQUIPPABLE component
        EquipmentSlot slot = this.getEquipmentSlotForItem(stack);
        if (slot == EquipmentSlot.MAINHAND) return false;       // không phải armor
        if (!this.canUseSlot(slot)) return false;
        if (!this.getItemBySlot(slot).isEmpty()) return false;  // slot đã có đồ

        // Mặc vào
        ItemStack toEquip = stack.copy();
        toEquip.setCount(1);
        this.setItemSlot(slot, toEquip);

        // Âm thanh mặc giáp (dùng sound của chính item nếu có)
        Equippable equippable = stack.get(DataComponents.EQUIPPABLE);
        if (equippable != null) {
            this.level().playSound(null, this.blockPosition(),
                    equippable.equipSound().value(),
                    SoundSource.NEUTRAL, 0.8f, 1.0f);
        } else {
            this.level().playSound(null, this.blockPosition(),
                    SoundEvents.ARMOR_EQUIP_GENERIC.value(),
                    SoundSource.NEUTRAL, 0.8f, 1.0f);
        }

        // Particle vui mắt
        if (this.level() instanceof ServerLevel serverLevel) {
            serverLevel.sendParticles(ParticleTypes.HAPPY_VILLAGER,
                    this.getX(), this.getY() + 0.5, this.getZ(),
                    5, 0.3, 0.3, 0.3, 0.1);
        }

        return true;
    }

    /** Kiểm tra còn slot armor nào trống không */
    public boolean hasEmptyArmorSlot() {
        for (EquipmentSlot slot : ARMOR_SLOTS) {
            if (this.getItemBySlot(slot).isEmpty()) {
                return true;
            }
        }
        return false;
    }

    @Override
    public InteractionResult mobInteract(Player player, InteractionHand hand) {
        ItemStack itemstack = player.getItemInHand(hand);

        // ========================================
        // 1. VẮT SỮA BẰNG XÔ
        // ========================================
        if (itemstack.is(Items.BUCKET) && this.isTame()) {
            ItemStack milkBucket = new ItemStack(Items.MILK_BUCKET);

            if (player.isCreative()) {
                if (!player.getInventory().add(milkBucket)) {
                    player.drop(milkBucket, false);
                }
            } else {
                if (itemstack.getCount() == 1) {
                    player.setItemInHand(hand, milkBucket);
                } else {
                    itemstack.shrink(1);
                    if (!player.getInventory().add(milkBucket)) {
                        player.drop(milkBucket, false);
                    }
                }
            }

            this.level().playSound(null, this.blockPosition(),
                    SoundEvents.COW_MILK, SoundSource.NEUTRAL, 1.0F, 1.0F);

            return InteractionResult.SUCCESS;
        }

        // ========================================
        // 2. LÚA MÌ
        // ========================================
        if (itemstack.is(Items.WHEAT)) {
            if (!this.isTame()) {
                if (!player.isCreative()) itemstack.shrink(1);
                if (this.random.nextInt(3) == 0) {
                    this.tame(player);
                    this.navigation.stop();
                    this.setTarget(null);
                    this.level().broadcastEntityEvent(this, (byte) 7);
                } else {
                    this.level().broadcastEntityEvent(this, (byte) 6);
                }
                return InteractionResult.SUCCESS;
            } else if (this.getHealth() < this.getMaxHealth()) {
                if (!player.isCreative()) itemstack.shrink(1);
                this.heal(4.0F);
                this.level().playSound(null, this.blockPosition(),
                        SoundEvents.GENERIC_EAT.value(), this.getSoundSource(), 1.0F, 1.0F);
                if (this.level() instanceof ServerLevel serverLevel) {
                    serverLevel.sendParticles(ParticleTypes.HEART,
                            this.getX(), this.getY() + 0.5, this.getZ(),
                            3, 0.5, 0.5, 0.5, 0.1);
                }
                return InteractionResult.SUCCESS;
            } else if (this.canFallInLove() && this.isTame()) {
                if (!player.isCreative()) itemstack.shrink(1);
                this.setInLove(player);
                this.level().playSound(null, this.blockPosition(),
                        SoundEvents.GENERIC_EAT.value(), this.getSoundSource(), 1.0F, 1.0F);
                return InteractionResult.SUCCESS;
            }
        }

        // ========================================
        // 3. TAY KHÔNG + LÀ CHỦ
        // ========================================
        if (this.isTame() && itemstack.isEmpty() && this.isOwnedBy(player)) {
            // Shift → thả item đang cầm
            if (player.isShiftKeyDown()) {
                ItemStack held = this.getItemBySlot(EquipmentSlot.MAINHAND);
                if (!held.isEmpty()) {
                    if (this.level() instanceof ServerLevel serverLevel) {
                        this.spawnAtLocation(serverLevel, held);
                    }
                    this.setItemSlot(EquipmentSlot.MAINHAND, ItemStack.EMPTY);
                    this.level().playSound(null, this.blockPosition(),
                            SoundEvents.ITEM_PICKUP, SoundSource.NEUTRAL, 0.5f, 1.0f);
                    return InteractionResult.SUCCESS;
                }
            }

            // Click thường → ngồi/đứng
            this.setOrderedToSit(!this.isOrderedToSit());
            this.jumping = false;
            this.navigation.stop();
            this.setTarget(null);
            return InteractionResult.SUCCESS;
        }

        return super.mobInteract(player, hand);
    }

    // CUSTOM GOAL: Tìm item → thử mặc giáp hoặc cầm tay
    // ============================================
    public static class SeekItemGoal extends Goal {
        private final BabyBat bat;
        private final double speedModifier;
        private final int searchRange;
        private ItemEntity targetItem;
        private int cooldown;

        public SeekItemGoal(BabyBat bat, double speed, int range) {
            this.bat = bat;
            this.speedModifier = speed;
            this.searchRange = range;
            this.setFlags(EnumSet.of(Goal.Flag.MOVE, Goal.Flag.LOOK));
        }

        @Override
        public boolean canUse() {
            if (this.cooldown > 0) {
                this.cooldown--;
                return false;
            }
            if (this.bat.isOrderedToSit()) return false;

            // 👈 Chỉ dừng khi CẢ mainhand + armor đều đầy
            boolean mainhandFull = !this.bat.getItemBySlot(EquipmentSlot.MAINHAND).isEmpty();
            boolean armorFull = !this.bat.hasEmptyArmorSlot();
            if (mainhandFull && armorFull) return false;

            List<ItemEntity> items = this.bat.level().getEntitiesOfClass(
                    ItemEntity.class,
                    this.bat.getBoundingBox().inflate(this.searchRange),
                    item -> !item.isRemoved() && !item.getItem().isEmpty()
            );
            if (items.isEmpty()) return false;

            items.sort(Comparator.comparingDouble(this.bat::distanceToSqr));
            this.targetItem = items.get(0);
            return true;
        }

        @Override
        public void start() {
            if (this.targetItem != null) {
                this.bat.getNavigation().moveTo(this.targetItem, this.speedModifier);
            }
        }

        @Override
        public boolean canContinueToUse() {
            return this.targetItem != null
                    && !this.targetItem.isRemoved()
                    && this.bat.distanceToSqr(this.targetItem) < (double) (this.searchRange * this.searchRange);
        }

        @Override
        public void tick() {
            if (this.targetItem == null || this.targetItem.isRemoved()) return;

            this.bat.getLookControl().setLookAt(this.targetItem);

            if (this.bat.distanceToSqr(this.targetItem) < 4.0) {
                this.bat.getNavigation().stop();

                ItemStack stack = this.targetItem.getItem();
                if (stack.isEmpty()) return;

                // 👈 Ưu tiên 1: Thử mặc giáp
                if (this.bat.tryEquipArmor(stack)) {
                    stack.shrink(1);
                    if (stack.isEmpty()) {
                        this.targetItem.discard();
                    } else {
                        this.targetItem.setItem(stack);
                    }
                    return;
                }

                // 👈 Ưu tiên 2: Cầm tay (nếu tay trống)
                if (this.bat.getItemBySlot(EquipmentSlot.MAINHAND).isEmpty()) {
                    ItemStack toHold = stack.copy();
                    toHold.setCount(1);
                    this.bat.setItemSlot(EquipmentSlot.MAINHAND, toHold);

                    stack.shrink(1);
                    if (stack.isEmpty()) {
                        this.targetItem.discard();
                    } else {
                        this.targetItem.setItem(stack);
                    }

                    this.bat.level().playSound(null, this.bat.blockPosition(),
                            SoundEvents.ITEM_PICKUP, SoundSource.NEUTRAL, 0.5f, 1.0f);

                    if (this.bat.level() instanceof ServerLevel serverLevel) {
                        serverLevel.sendParticles(ParticleTypes.HEART,
                                this.bat.getX(), this.bat.getY() + 0.5, this.bat.getZ(),
                                3, 0.3, 0.3, 0.3, 0.1);
                    }
                }
            } else {
                this.bat.getNavigation().moveTo(this.targetItem, this.speedModifier);
            }
        }

        @Override
        public void stop() {
            this.targetItem = null;
            this.cooldown = 20;
        }
    }
}.

/**
 * Bot player cho Minecraft 26.2 (Fabric).
 *
 * Các lỗi của bản cũ đã sửa:
 *  1. tick() không gọi super.tick() / doTick() -> entity không được cập nhật (vật lý, trang bị, hồi chiêu đánh...).
 *     FakeClientConnection.tick() rỗng nên doTick() cũng không bao giờ được gọi ở nơi khác.
 *  2. Di chuyển phụ thuộc vào zza (bị hệ thống input của ServerPlayer ghi đè) -> giờ đặt vận tốc trực tiếp.
 *  3. Tấn công dùng ATTACK_DAMAGE (=1) thay vì Player.attack(), và đang cầm khối đá thay vì kiếm
 *     -> giờ tự chọn kiếm/rìu và gọi attack() đúng chuẩn.
 *  4. BUILD_TOWER chỉ chạy khi đứng trên đất và dưới chân trống (không bao giờ xảy ra) -> giờ nhảy rồi đặt block.
 *  5. BUILD_BRIDGE không ngồi (sneak) nên rơi khỏi mép -> giờ sneak và đặt block ở phía trước.
 *  6. PATROL đảo hướng mỗi tick khi ra xa >10 block (rung lắc) -> giờ chỉ đảo khi còn đang đi ra xa.
 *  7. FOLLOW/FLEE dùng getNearestPlayer có thể chọn nhầm bot khác -> giờ theo chủ, hoặc người chơi thật gần nhất.
 *  8. MINER đào thẳng xuống rồi rơi -> giờ đào hầm ngang, dừng/quay đầu khi gặp chất lỏng hoặc hố.
 *  9. Bỏ println debug, thêm xử lý khi bot chết.
 */
public class BotPlayer extends FakePlayer {

    public enum Mode {
        IDLE, WANDER, FOLLOW, FLEE, ATTACK, PVP, GUARD, PATROL,
        BUILD_BRIDGE, BUILD_TOWER, MINER, JUMPER, DANCER
    }

    /** Hệ số đổi attribute MOVEMENT_SPEED (0.1) -> block/tick khi đi bộ (~0.216). */
    private static final double WALK_FACTOR = 2.16D;
    private static final double REACH = 3.0D;

    private Mode mode = Mode.WANDER;
    private UUID ownerUUID;

    private int wanderTimer = 0;
    private int wanderPause = 0;
    private int buildCooldown = 0;
    private int actionCooldown = 0;
    private int blockedTicks = 0;
    private int deadTicks = 0;
    private float patrolYaw = 0f;
    private BlockPos patrolCenter;

    // Trạng thái mong muốn trong tick hiện tại
    private boolean wantSprint = false;
    private boolean wantSneak = false;

    public BotPlayer(ServerLevel level, GameProfile profile) {
        super(level, profile);
        this.setInvulnerable(false);
    }

    // ============================================
    // API
    // ============================================
    public void setMode(Mode mode) {
        this.mode = mode;
        this.patrolCenter = null;
        this.blockedTicks = 0;
        this.wanderTimer = 0;

        this.buildCooldown = 0;

    }

    public Mode getMode() { return this.mode; }

    // ============================================
    // Cho phép server chạy physics + AI
    // ============================================
    @Override
    public boolean isEffectiveAi() {
        return true;
    }

    @Override
    public boolean canSimulateMovement() {

    public float getSpeed() {

    // ============================================
    // TICK
    // ============================================
    @Override
    public void tick() {
        MinecraftServer server = this.level().getServer();
        if (server == null) return;

        if (!this.isAlive()) {
            // Bot không tự hồi sinh được -> gỡ khỏi server sau 2 giây
            if (++this.deadTicks == 40) {
                server.execute(() -> server.getPlayerList().remove(this));
            }
            return;
        }

        if (this.buildCooldown > 0) this.buildCooldown--;
        if (this.actionCooldown > 0) this.actionCooldown--;

        this.zza = 0f;
        this.xxa = 0f;
        this.yya = 0f;
        this.wantSprint = false;
        this.wantSneak = false;

        switch (this.mode) {
            case IDLE         -> { }
            case WANDER       -> this.tickWander();
            case FOLLOW       -> this.tickFollow();
            case FLEE         -> this.tickFlee();
            case ATTACK       -> this.tickAttack(false);
            case PVP          -> this.tickAttack(true);
            case GUARD        -> this.tickGuard();
            case PATROL       -> this.tickPatrol();
            case BUILD_BRIDGE -> this.tickBuildBridge();
            case BUILD_TOWER  -> this.tickBuildTower();
            case MINER        -> this.tickMiner();
            case JUMPER       -> this.tickJumper();
            case DANCER       -> this.tickDancer();
        }

        if (this.isSprinting() != this.wantSprint) this.setSprinting(this.wantSprint);
        if (this.isShiftKeyDown() != this.wantSneak) this.setShiftKeyDown(this.wantSneak);

        // Giống Carpet: tick entity rồi tự gọi doTick() vì connection giả không tick.
        try {
            super.tick();
            this.doTick();
        } catch (NullPointerException ignored) {
        }

        if (this.tickCount % 10 == 0) {
            ((ServerLevel) this.level()).getChunkSource().move(this);
        }
    }

    // ============================================
    // WANDER
    // ============================================
    private void tickWander() {
        if (this.wanderPause > 0) {
            this.wanderPause--;
            return;
        }
        if (--this.wanderTimer <= 0) {
            this.wanderTimer = 40 + this.random.nextInt(60);
            if (this.random.nextInt(4) == 0) {
                this.wanderPause = 20 + this.random.nextInt(30);
                return;
            }
            this.rotateTo(this.random.nextFloat() * 360f);
        }
        this.walkForward(1.0f);
        this.autoJump();
    }

    // ============================================
    // FOLLOW
    // ============================================
    private void tickFollow() {
        Player target = this.findFollowTarget(48.0);
        if (target == null) { this.tickWander(); return; }

        this.faceEntity(target);
        double dist = this.distanceTo(target);
        if (dist > 3.0) {
            this.wantSprint = dist > 8.0;
            this.walkForward(1.0f);
            this.autoJump();
        }
    }

    // ============================================
    // FLEE
    // ============================================
    private void tickFlee() {
        Player threat = this.nearestRealPlayer(16.0);
        if (threat == null) { this.tickWander(); return; }

        Vec3 away = this.position().subtract(threat.position());
        this.rotateTo((float) Math.toDegrees(Math.atan2(-away.x, away.z)));
        this.wantSprint = true;
        this.walkForward(1.0f);
        this.autoJump();
    }

    // ============================================
    // ATTACK / PVP
    // ============================================
    private void tickAttack(boolean pvp) {
        LivingEntity target = pvp ? this.findPlayerTarget(24.0) : this.findEnemyTarget(24.0);
        if (target == null) {
            // Không có mục tiêu -> đi theo chủ cho gọn
            if (this.findFollowTarget(48.0) != null) this.tickFollow();
            return;
        }

        this.selectHotbarWhere(s -> s.is(ItemTags.SWORDS));
        this.faceEntity(target);

        double reachSqr = REACH * REACH;
        boolean inReach = target.getBoundingBox().distanceToSqr(this.getEyePosition()) <= reachSqr;

        if (inReach) {
            this.tryHit(target);
        } else {
            this.wantSprint = this.distanceTo(target) > 5.0;
            this.walkForward(1.0f);
            this.autoJump();
        }
    }

    // ============================================
    // GUARD
    // ============================================
    private void tickGuard() {
        LivingEntity target = this.findEnemyTarget(8.0);
        if (target == null) return;

        this.faceEntity(target);
        if (target.getBoundingBox().distanceToSqr(this.getEyePosition()) <= REACH * REACH) {
            this.selectHotbarWhere(s -> s.is(ItemTags.SWORDS));
            this.tryHit(target);
        }
    }

    // ============================================
    // PATROL
    // ============================================
    private void tickPatrol() {
        if (this.patrolCenter == null) {
            this.patrolCenter = this.blockPosition();
            this.patrolYaw = this.getYRot();
        }

        double ox = this.getX() - (this.patrolCenter.getX() + 0.5);
        double oz = this.getZ() - (this.patrolCenter.getZ() + 0.5);
        double dist = Math.sqrt(ox * ox + oz * oz);

        // Hướng đang đi (theo yaw của Minecraft)
        double rad = Math.toRadians(this.patrolYaw);
        double fx = -Math.sin(rad);
        double fz = Math.cos(rad);
        boolean movingAway = (ox * fx + oz * fz) > 0;

        if (dist > 10.0 && movingAway) {
            this.patrolYaw += 180f + (this.random.nextFloat() - 0.5f) * 40f;
        }

        if (this.horizontalCollision) {
            if (++this.blockedTicks > 15) {
                this.patrolYaw += 90f + this.random.nextInt(90);
                this.blockedTicks = 0;
            }
        } else {
            this.blockedTicks = 0;
        }

        this.rotateTo(this.patrolYaw);
        this.walkForward(0.7f);
        this.autoJump();
    }

    // ============================================
    // BUILD_BRIDGE (đi + sneak + đặt block phía trước dưới chân)
    // ============================================
    private void tickBuildBridge() {
        this.wantSneak = true;
        this.autoJump();

        Direction dir = this.getDirection();
        BlockPos under = this.blockPosition().below();
        BlockPos frontUnder = under.relative(dir);

        if (this.buildCooldown <= 0) {
            if (this.canPlaceAt(under)) {
                if (this.tryPlaceBlock(under)) this.buildCooldown = 3;
            } else if (this.canPlaceAt(frontUnder)) {
                if (this.tryPlaceBlock(frontUnder)) this.buildCooldown = 3;
            }
        }

        // Chỉ bước tới khi phía trước đã có nền
        if (!this.canPlaceAt(frontUnder)) {
            this.walkForward(1.0f);
        }
    }

    // ============================================
    // BUILD_TOWER (nhảy rồi đặt block dưới chân)
    // ============================================
    private void tickBuildTower() {
        if (this.blockPosition().getY() >= this.level().getMaxY() - 2) return;

        if (this.onGround()) {
            this.jumpFromGround();
            return;
        }

        BlockPos below = this.blockPosition().below();
        // Khi đã nhảy lên đủ cao, ô dưới chân (nơi vừa đứng) sẽ trống -> đặt block vào đó
        if (this.buildCooldown <= 0 && this.getY() >= below.getY() + 1.0 && this.canPlaceAt(below)) {
            if (this.tryPlaceBlock(below)) this.buildCooldown = 3;
        }
    }

    // ============================================
    // MINER (đào hầm ngang theo hướng đang nhìn)
    // ============================================
    private void tickMiner() {
        this.selectHotbarWhere(s -> s.is(ItemTags.PICKAXES));

        Direction dir = this.getDirection();
        this.rotateTo(dir.toYRot());

        BlockPos front = this.blockPosition().relative(dir);
        BlockPos[] hazards = { front, front.above(), front.below() };
        for (BlockPos p : hazards) {
            if (!this.level().getFluidState(p).isEmpty()) {
                this.rotateTo(this.getYRot() + 90f);
                return;
            }
        }

        BlockPos[] column = { front, front.above() };
        for (BlockPos pos : column) {
            BlockState state = this.level().getBlockState(pos);
            if (state.isAir()) continue;

            if (state.getDestroySpeed(this.level(), pos) < 0) { // bedrock...
                this.rotateTo(this.getYRot() + 90f);
                return;
            }
            if (this.actionCooldown > 0) return;

            this.gameMode.destroyBlock(pos);
            this.swing(InteractionHand.MAIN_HAND);
            this.actionCooldown = 5;
            return;
        }

        // Đường thông -> kiểm tra có nền để đi tiếp không
        if (this.level().getBlockState(front.below()).isAir()) {
            this.rotateTo(this.getYRot() + 90f);
            return;
        }
        this.walkForward(0.8f);
    }

    // ============================================
    // JUMPER
    // ============================================
    private void tickJumper() {
        if (this.onGround()) this.jumpFromGround();
    }

    // ============================================
    // DANCER
    // ============================================
    private void tickDancer() {
        if (this.tickCount % 5 == 0) this.rotateTo(this.getYRot() + 45f);
        if (this.onGround() && this.random.nextInt(10) == 0) this.jumpFromGround();
        this.wantSneak = (this.tickCount / 8) % 2 == 0;
        if (this.random.nextInt(10) == 0) {
            this.swing(this.random.nextBoolean() ? InteractionHand.MAIN_HAND : InteractionHand.OFF_HAND);
        }
    }

    // ============================================
    // MOVEMENT HELPERS
    // ============================================

    /** Đi theo hướng yaw hiện tại bằng cách đặt thẳng vận tốc ngang (giữ nguyên vận tốc dọc). */
    private void walkForward(float factor) {
        double speed = this.getAttributeValue(Attributes.MOVEMENT_SPEED) * WALK_FACTOR * factor;
        if (this.wantSneak) speed *= 0.3D;

        float yaw = this.getYRot() * Mth.DEG_TO_RAD;
        Vec3 d = this.getDeltaMovement();
        double dy = d.y;
        if (this.isInWater()) {
            speed *= 0.6D;
            dy = Math.min(dy + 0.04D, 0.2D); // nổi lên
        }
        this.setDeltaMovement(-Mth.sin(yaw) * speed, dy, Mth.cos(yaw) * speed);
    }

    private void autoJump() {
        if (this.horizontalCollision && this.onGround()) this.jumpFromGround();
    }

    // ============================================
    // COMBAT HELPERS
    // ============================================
    private void tryHit(LivingEntity target) {
        if (this.getAttackStrengthScale(0.5F) < 1.0F) return;
        this.swing(InteractionHand.MAIN_HAND);
        this.attack(target); // xử lý vũ khí, enchant, knockback, crit chuẩn vanilla
    }

    private LivingEntity findEnemyTarget(double range) {
        return this.level().getEntitiesOfClass(LivingEntity.class,
                        this.getBoundingBox().inflate(range),
                        e -> e != this && e.isAlive() && e instanceof Enemy)
                .stream()
                .min(Comparator.comparingDouble(this::distanceToSqr))
                .orElse(null);
    }

    private LivingEntity findPlayerTarget(double range) {
        return this.level().getEntitiesOfClass(Player.class,
                        this.getBoundingBox().inflate(range),
                        p -> p != this && p.isAlive() && !p.isCreative() && !p.isSpectator()
                                && !(p instanceof BotPlayer)
                                && (this.ownerUUID == null || !p.getUUID().equals(this.ownerUUID)))
                .stream()
                .min(Comparator.comparingDouble(this::distanceToSqr))
                .orElse(null);
    }

    private Player findFollowTarget(double range) {
        if (this.ownerUUID != null) {
            Player owner = this.level().getPlayerByUUID(this.ownerUUID);
            if (owner != null && owner.isAlive() && !owner.isSpectator()
                    && owner.level() == this.level()) {
                return owner;
            }
        }
        return this.nearestRealPlayer(range);
    }

    private Player nearestRealPlayer(double range) {
        Player best = null;
        double bestDist = range * range;
        for (Player p : this.level().players()) {
            if (p == this || p instanceof BotPlayer || !p.isAlive() || p.isSpectator()) continue;
            double d = this.distanceToSqr(p);
            if (d < bestDist) {
                bestDist = d;
                best = p;
            }
        }
        return best;
    }

    // ============================================
    // INVENTORY HELPERS
    // ============================================
    private boolean selectHotbarWhere(Predicate<ItemStack> predicate) {
        Inventory inv = this.getInventory();
        if (predicate.test(inv.getItem(inv.getSelectedSlot()))) return true;
        for (int i = 0; i < 9; i++) {
            if (predicate.test(inv.getItem(i))) {
                inv.setSelectedSlot(i);
                return true;
            }
        }
        return false;
    }

    // ============================================
    // BUILD HELPERS
    // ============================================
    private boolean canPlaceAt(BlockPos pos) {
        return this.level().getBlockState(pos).canBeReplaced();
    }

    private ItemStack findPlaceableStack() {
        Inventory inv = this.getInventory();
        for (int i = 0; i < 36; i++) {
            ItemStack stack = inv.getItem(i);
            if (!stack.isEmpty() && stack.getItem() instanceof BlockItem bi && isPlaceable(bi)) {
                if (i < 9) inv.setSelectedSlot(i);
                return stack;
            }
        }
        return ItemStack.EMPTY;
    }

    private static boolean isPlaceable(BlockItem item) {
        var block = item.getBlock();
        return block != Blocks.AIR
                && !(block instanceof FallingBlock)
                && !block.defaultBlockState().hasBlockEntity();
    }

    private boolean tryPlaceBlock(BlockPos pos) {
        ItemStack stack = this.findPlaceableStack();
        if (stack.isEmpty()) return false;

        BlockItem blockItem = (BlockItem) stack.getItem();
        BlockState state = blockItem.getBlock().defaultBlockState();
        if (!state.canSurvive(this.level(), pos)) return false;

        if (this.level().setBlock(pos, state, 3)) {
            this.level().playSound(null, pos, state.getSoundType().getPlaceSound(),
                    SoundSource.BLOCKS, 1.0f, 1.0f);
            if (!this.isCreative()) stack.shrink(1);
            this.swing(InteractionHand.MAIN_HAND);
            return true;
        }
        return false;
    }

    // ============================================
    // ROTATION HELPERS
    // ============================================
    private void faceEntity(Entity target) {
        Vec3 diff = target.getEyePosition().subtract(this.getEyePosition());
        double horiz = Math.sqrt(diff.x * diff.x + diff.z * diff.z);
        this.rotateTo((float) Math.toDegrees(Math.atan2(-diff.x, diff.z)));
        this.setXRot((float) -Math.toDegrees(Math.atan2(diff.y, horiz)));
    }

    private void rotateTo(float yaw) {
        this.setYRot(yaw);
        this.setYHeadRot(yaw);
        this.setYBodyRot(yaw);
    }
}.

                .add(Attributes.FOLLOW_RANGE, 200.0D)
                .add(Attributes.ARMOR, 20.0D);

        this.goalSelector.addGoal(0, new FloatGoal(this));

        this.goalSelector.addGoal(3, new LookAtPlayerGoal(this, Player.class, 8.0F));
        this.goalSelector.addGoal(4, new RandomLookAroundGoal(this));

        this.targetSelector.addGoal(2, new NearestAttackableTargetGoal<>(this, Player.class, true));

        super.stopSeenByPlayer(player);

    @Override
    public void aiStep() {
        super.aiStep();
        
        // 1. Cập nhật thanh tiến trình (độ dài của thanh máu)
        float healthRatio = this.getHealth() / this.getMaxHealth();
        this.bossBar.setProgress(healthRatio);

        // 2. Tự động đổi màu thanh máu dựa trên % máu
        if (healthRatio > 0.66F) {
            // Trên 66% máu: Màu Xanh Lá (Khỏe mạnh)
            this.bossBar.setColor(BossEvent.BossBarColor.GREEN);
        } else if (healthRatio > 0.33F) {
            // Từ 33% đến 66% máu: Màu Vàng (Bị thương)
            this.bossBar.setColor(BossEvent.BossBarColor.YELLOW);
        } else {
            // Dưới 33% máu: Màu Đỏ (Nguy kịch)
            this.bossBar.setColor(BossEvent.BossBarColor.RED);
        }

        // 3. Logic kỹ năng
        if (!this.level().isClientSide() && this.isAlive()) {
            LivingEntity target = this.getTarget();
            
            if (target != null) {
                if (this.skillCooldown > 0) {
                    this.skillCooldown--;
                } else {
                    useRandomSkill(target);
                    this.skillCooldown = 200; // Hồi chiêu 10 giây
                }
            }
        }
    }

    private void useRandomSkill(LivingEntity target) {
        // Tăng lên 5 kỹ năng (từ 0 đến 4)
        int skill = this.getRandom().nextInt(5);
        switch (skill) {
            case 0 -> teleportBehindTarget(target);
            case 1 -> summonLightning(target);
            case 2 -> curseOfDarkness(target);
            case 3 -> summonUndead(target);
            case 4 -> fearKnockback(target);
        }
    }

        double y = target.getY();

        if (this.level() instanceof ServerLevel serverLevel) {
            serverLevel.sendParticles(ParticleTypes.PORTAL, this.getX(), this.getY() + 1, this.getZ(), 30, 0.5, 0.5, 0.5, 0.1);

            this.level().playSound(null, this.blockPosition(), SoundEvents.ENDERMAN_SCREAM, SoundSource.HOSTILE, 1.0f, 0.5f);
            
            this.swing(InteractionHand.MAIN_HAND);
            this.doHurtTarget(serverLevel, target);

                double dz = this.getRandom().nextDouble() * 4 - 2;
                
                Zombie zombie = EntityTypes.ZOMBIE.spawn(serverLevel, 
                        this.blockPosition().offset((int)dx, 0, (int)dz), 
                        EntitySpawnReason.TRIGGERED);

                    serverLevel.sendParticles(ParticleTypes.SMOKE, zombie.getX(), zombie.getY() + 1, zombie.getZ(), 20, 0.5, 0.5, 0.5, 0.1);

        this.level().playSound(null, this.blockPosition(), SoundEvents.WARDEN_ROAR, SoundSource.HOSTILE, 1.0f, 0.8f);
        
        // Tính toán hướng đẩy lùi từ Herobrine đến mục tiêu
        Vec3 pushVec = target.position().subtract(this.position()).normalize();
        
        // Đẩy lùi mục tiêu ra xa (tốc độ X/Z, độ cao Y)
        target.push(pushVec.x * 1.5, 0.6, pushVec.z * 1.5);
        
        // Gây sát thương nhẹ (2 máu)
        if (this.level() instanceof ServerLevel serverLevel) {
            target.hurtServer(serverLevel, this.damageSources().mobAttack(this), 2.0f);

            serverLevel.sendParticles(ParticleTypes.SONIC_BOOM, this.getX(), this.getY() + 1, this.getZ(), 1, 0, 0, 0, 0);

                        Component.translatable("message.igoo.boss.defeated_unknown", 
                                this.getDisplayName()),
                        false
                );
            }
            
            // (Tùy chọn) Phát âm thanh chiến thắng
            this.level().playSound(null, this.blockPosition(), net.minecraft.sounds.SoundEvents.UI_TOAST_CHALLENGE_COMPLETE, net.minecraft.sounds.SoundSource.MASTER, 1.0f, 1.0f);
        }
    }

    private static final Set<BlockPos> ACTIVE_ALTARS = new HashSet<>();

    public static void register() {
        UseBlockCallback.EVENT.register((player, level, hand, hitResult) -> {
            if (level.isClientSide() || hand != InteractionHand.MAIN_HAND) {
                return InteractionResult.PASS;
            }

            ItemStack stack = player.getItemInHand(hand);
            BlockPos pos = hitResult.getBlockPos();

            if (stack.is(Items.FLINT_AND_STEEL) && level.getBlockState(pos).is(Blocks.NETHERRACK)) {
                if (ACTIVE_ALTARS.contains(pos)) {
                    return InteractionResult.FAIL;
                }

                if (isValidAltar(level, pos) && level instanceof ServerLevel serverLevel) {
                    ACTIVE_ALTARS.add(pos);

                    level.setBlockAndUpdate(pos.above(), Blocks.FIRE.defaultBlockState());
                    level.playSound(null, pos, SoundEvents.WITHER_AMBIENT, SoundSource.HOSTILE, 1.0f, 1.0f);
                    player.sendSystemMessage(Component.translatable("message.igoo.herobrine.summoning"));

                    if (!player.isCreative()) {
                        stack.hurtAndBreak(1, player, EquipmentSlot.MAINHAND);
                    }

                    CompletableFuture.delayedExecutor(5, TimeUnit.SECONDS).execute(() -> {
                        serverLevel.getServer().execute(() -> {
                            ACTIVE_ALTARS.remove(pos);
                            destroyAltar(level, pos);

                            ResourceKey<Biome> cursedBiomeKey = ResourceKey.create(
                                    Registries.BIOME,
                                    Identifier.fromNamespaceAndPath(Igoo.MOD_ID, "cursed_savanna"));
                            boolean isCursed = level.getBiome(pos).is(cursedBiomeKey);

                            HerobrineBoss boss = isCursed
                                    ? ModEntities.CURSED_HEROBRINE.spawn(serverLevel, pos.above(),
                                            net.minecraft.world.entity.EntitySpawnReason.TRIGGERED)
                                    : ModEntities.HEROBRINE.spawn(serverLevel, pos.above(),
                                            net.minecraft.world.entity.EntitySpawnReason.TRIGGERED);

                            if (boss != null) {
                                if (isCursed) {
                                    level.playSound(null, pos, SoundEvents.WARDEN_EMERGE,
                                            SoundSource.HOSTILE, 1.0f, 0.5f);
                                    player.sendSystemMessage(Component.translatable(
                                            "message.igoo.herobrine.cursed_summoned"));
                                } else {
                                    level.playSound(null, pos, SoundEvents.WITHER_SPAWN,
                                            SoundSource.HOSTILE, 1.0f, 1.0f);
                                    player.sendSystemMessage(Component.translatable(
                                            "message.igoo.herobrine.summoned"));
                                }
                            }
                        });
                    });

                    return InteractionResult.SUCCESS;
                }
            }
            return InteractionResult.PASS;
        });
    }

    private static void destroyAltar(Level level, BlockPos center) {
        level.removeBlock(center.above(), false);
        level.removeBlock(center, false);

        BlockPos goldCenter = center.below();
        for (int x = -1; x <= 1; x++) {
            for (int z = -1; z <= 1; z++) {
                level.removeBlock(goldCenter.offset(x, 0, z), false);
            }
        }

        BlockPos[] torchPositions = { center.north(), center.south(), center.east(), center.west() };
        for (BlockPos torchPos : torchPositions) {
            level.removeBlock(torchPos, false);
        }

        level.playSound(null, center, SoundEvents.GENERIC_EXPLODE.value(), SoundSource.BLOCKS, 1.0f, 1.0f);
        if (level instanceof ServerLevel serverLevel) {
            serverLevel.sendParticles(ParticleTypes.EXPLOSION,
                    center.getX() + 0.5, center.getY() + 0.5, center.getZ() + 0.5,
                    20, 1.5, 1.5, 1.5, 0.1);
            serverLevel.sendParticles(ParticleTypes.LARGE_SMOKE,
                    center.getX() + 0.5, center.getY() + 0.5, center.getZ() + 0.5,
                    30, 1.0, 1.0, 1.0, 0.05);
        }
    }

    private static boolean isValidAltar(Level level, BlockPos center) {
        BlockPos goldCenter = center.below();
        for (int x = -1; x <= 1; x++) {
            for (int z = -1; z <= 1; z++) {
                if (!level.getBlockState(goldCenter.offset(x, 0, z)).is(Blocks.GOLD_BLOCK)) {
                    return false;
                }
            }
        }

        BlockPos[] torchPositions = { center.north(), center.south(), center.east(), center.west() };
        for (BlockPos torchPos : torchPositions) {
            var state = level.getBlockState(torchPos);
            if (!state.is(Blocks.REDSTONE_TORCH) && !state.is(Blocks.REDSTONE_WALL_TORCH)) {
                return false;
            }
        }
        return true;
    }
}.

    public static final ResourceKey<ConfiguredFeature<?, ?>> ALUMINUM_ORE_KEY = ResourceKey.create(
            Registries.CONFIGURED_FEATURE, Igoo.id("aluminum_ore"));
    public static final ResourceKey<ConfiguredFeature<?, ?>> TIN_ORE_KEY = ResourceKey.create(
            Registries.CONFIGURED_FEATURE, Igoo.id("tin_ore"));
    public static final ResourceKey<ConfiguredFeature<?, ?>> CHROMIUM_ORE_KEY = ResourceKey.create(
            Registries.CONFIGURED_FEATURE, Igoo.id("chromium_ore"));
    public static final ResourceKey<ConfiguredFeature<?, ?>> TITANIUM_ORE_KEY = ResourceKey.create(
            Registries.CONFIGURED_FEATURE, Igoo.id("titanium_ore"));
    public static final ResourceKey<ConfiguredFeature<?, ?>> TUNGSTEN_ORE_KEY = ResourceKey.create(
            Registries.CONFIGURED_FEATURE, Igoo.id("tungsten_ore"));

    public static void configure(BootstrapContext<ConfiguredFeature<?, ?>> context) {
        // Quy tắc: Quặng có thể thay thế Đá (Stone) và Đá sâu (Deepslate)
        var stoneReplaceable = new TagMatchTest(BlockTags.STONE_ORE_REPLACEABLES);

        // --- Cấu hình cho Nhôm (Mạch 9 khối) ---
        List<OreConfiguration.TargetBlockState> aluminumTargets = List.of(
                OreConfiguration.target(stoneReplaceable, ModBlocks.Aluminum_ore.defaultBlockState())
        );
        context.register(ALUMINUM_ORE_KEY, new ConfiguredFeature<>(Feature.ORE, new OreConfiguration(aluminumTargets, 9)));

        // --- Cấu hình cho Thiếc (Mạch 8 khối) ---
        List<OreConfiguration.TargetBlockState> tinTargets = List.of(
                OreConfiguration.target(stoneReplaceable, ModBlocks.Tin_ore.defaultBlockState())
        );
        context.register(TIN_ORE_KEY, new ConfiguredFeature<>(Feature.ORE, new OreConfiguration(tinTargets, 8)));

        // --- Cấu hình cho Crom (Mạch 6 khối) ---
        List<OreConfiguration.TargetBlockState> chromiumTargets = List.of(
                OreConfiguration.target(stoneReplaceable, ModBlocks.Chromium_ore.defaultBlockState())
        );
        context.register(CHROMIUM_ORE_KEY, new ConfiguredFeature<>(Feature.ORE, new OreConfiguration(chromiumTargets, 6)));

        // --- Cấu hình cho Titan (Mạch 5 khối) ---
        List<OreConfiguration.TargetBlockState> titaniumTargets = List.of(
                OreConfiguration.target(stoneReplaceable, ModBlocks.Titanium_ore.defaultBlockState())
        );
        context.register(TITANIUM_ORE_KEY, new ConfiguredFeature<>(Feature.ORE, new OreConfiguration(titaniumTargets, 5)));

        // --- Cấu hình cho Vonfram (Mạch 4 khối) ---
        List<OreConfiguration.TargetBlockState> tungstenTargets = List.of(
                OreConfiguration.target(stoneReplaceable, ModBlocks.Tungsten_ore.defaultBlockState())
        );
        context.register(TUNGSTEN_ORE_KEY, new ConfiguredFeature<>(Feature.ORE, new OreConfiguration(tungstenTargets, 4)));
    }
}.

package com.igoo.worldgen;

import com.igoo.Igoo;
import net.minecraft.core.HolderGetter;
import net.minecraft.core.registries.Registries;
import net.minecraft.data.worldgen.BootstrapContext;
import net.minecraft.data.worldgen.features.OreFeatures;
import net.minecraft.data.worldgen.features.VegetationFeatures;
import net.minecraft.resources.Identifier;
import net.minecraft.resources.ResourceKey;
import net.minecraft.world.level.levelgen.VerticalAnchor;
import net.minecraft.world.level.levelgen.feature.ConfiguredFeature;
import net.minecraft.world.level.levelgen.heightproviders.TrapezoidHeight;
import net.minecraft.world.level.levelgen.placement.*;

import java.util.List;
import net.minecraft.data.worldgen.placement.PlacementUtils;

public class ModWorldPlacedFeatures {

    // ===== 1. Key quặng của Mod =====
    public static final ResourceKey<PlacedFeature> ALUMINUM_ORE_PLACED_KEY = key("aluminum_ore_placed");
    public static final ResourceKey<PlacedFeature> TIN_ORE_PLACED_KEY = key("tin_ore_placed");
    public static final ResourceKey<PlacedFeature> CHROMIUM_ORE_PLACED_KEY = key("chromium_ore_placed");
    public static final ResourceKey<PlacedFeature> TITANIUM_ORE_PLACED_KEY = key("titanium_ore_placed");
    public static final ResourceKey<PlacedFeature> TUNGSTEN_ORE_PLACED_KEY = key("tungsten_ore_placed");

    // ===== 2. Key quặng Vanilla =====
    public static final ResourceKey<PlacedFeature> CURSED_ORE_COAL_UPPER = key("cursed_ore_coal_upper");
    public static final ResourceKey<PlacedFeature> CURSED_ORE_COAL_LOWER = key("cursed_ore_coal_lower");
    public static final ResourceKey<PlacedFeature> CURSED_ORE_IRON_UPPER = key("cursed_ore_iron_upper");
    public static final ResourceKey<PlacedFeature> CURSED_ORE_IRON_MIDDLE = key("cursed_ore_iron_middle");
    public static final ResourceKey<PlacedFeature> CURSED_ORE_IRON_SMALL = key("cursed_ore_iron_small");
    public static final ResourceKey<PlacedFeature> CURSED_ORE_COPPER = key("cursed_ore_copper");
    public static final ResourceKey<PlacedFeature> CURSED_ORE_GOLD = key("cursed_ore_gold");
    public static final ResourceKey<PlacedFeature> CURSED_ORE_REDSTONE = key("cursed_ore_redstone");
    public static final ResourceKey<PlacedFeature> CURSED_ORE_LAPIS = key("cursed_ore_lapis");
    
    // Biến thể Kim Cương
    public static final ResourceKey<PlacedFeature> CURSED_ORE_DIAMOND_SMALL = key("cursed_ore_diamond_small");
    public static final ResourceKey<PlacedFeature> CURSED_ORE_DIAMOND_MEDIUM = key("cursed_ore_diamond_medium");
    public static final ResourceKey<PlacedFeature> CURSED_ORE_DIAMOND_LARGE = key("cursed_ore_diamond_large");
    public static final ResourceKey<PlacedFeature> CURSED_ORE_DIAMOND_BURIED = key("cursed_ore_diamond_buried");

    // ===== 3. Key thảm thực vật =====
    public static final ResourceKey<PlacedFeature> CURSED_TREES_SAVANNA = key("cursed_trees_savanna");
    public static final ResourceKey<PlacedFeature> CURSED_PATCH_TALL_GRASS = key("cursed_patch_tall_grass");
    public static final ResourceKey<PlacedFeature> CURSED_PATCH_DEAD_BUSH = key("cursed_patch_dead_bush");

    private static ResourceKey<PlacedFeature> key(String name) {
        return ResourceKey.create(Registries.PLACED_FEATURE, Identifier.fromNamespaceAndPath(Igoo.MOD_ID, name));
    }

    public static void configure(BootstrapContext<PlacedFeature> context) {
        var cf = context.lookup(Registries.CONFIGURED_FEATURE);

        // ==========================================
        // A. QUẶNG CỦA MOD
        // ==========================================
        registerOre(context, cf, ALUMINUM_ORE_PLACED_KEY, ModWorldConfiguredFeatures.ALUMINUM_ORE_KEY, 12, 48, 128);
        registerOre(context, cf, TIN_ORE_PLACED_KEY, ModWorldConfiguredFeatures.TIN_ORE_KEY, 10, 24, 96);
        registerOre(context, cf, CHROMIUM_ORE_PLACED_KEY, ModWorldConfiguredFeatures.CHROMIUM_ORE_KEY, 6, 8, 48);
        registerOre(context, cf, TITANIUM_ORE_PLACED_KEY, ModWorldConfiguredFeatures.TITANIUM_ORE_KEY, 6, 4, 24);
        registerOre(context, cf, TUNGSTEN_ORE_PLACED_KEY, ModWorldConfiguredFeatures.TUNGSTEN_ORE_KEY, 6, 0, 128);

        // ==========================================
        // B. QUẶNG VANILLA (Clone)
        // ==========================================
        
        // Than (Coal)
        context.register(CURSED_ORE_COAL_UPPER, new PlacedFeature(cf.getOrThrow(OreFeatures.ORE_COAL), 
            List.of(CountPlacement.of(30), InSquarePlacement.spread(), HeightRangePlacement.uniform(VerticalAnchor.absolute(136), VerticalAnchor.top()), BiomeFilter.biome())));
        context.register(CURSED_ORE_COAL_LOWER, new PlacedFeature(cf.getOrThrow(OreFeatures.ORE_COAL_BURIED), 
            List.of(CountPlacement.of(20), InSquarePlacement.spread(), HeightRangePlacement.triangle(VerticalAnchor.absolute(0), VerticalAnchor.absolute(192)), BiomeFilter.biome())));

        // Sắt (Iron)
        context.register(CURSED_ORE_IRON_UPPER, new PlacedFeature(cf.getOrThrow(OreFeatures.ORE_IRON), 
            List.of(CountPlacement.of(90), InSquarePlacement.spread(), HeightRangePlacement.triangle(VerticalAnchor.absolute(80), VerticalAnchor.absolute(384)), BiomeFilter.biome())));
        context.register(CURSED_ORE_IRON_MIDDLE, new PlacedFeature(cf.getOrThrow(OreFeatures.ORE_IRON), 
            List.of(CountPlacement.of(10), InSquarePlacement.spread(), HeightRangePlacement.triangle(VerticalAnchor.absolute(-24), VerticalAnchor.absolute(56)), BiomeFilter.biome())));
        context.register(CURSED_ORE_IRON_SMALL, new PlacedFeature(cf.getOrThrow(OreFeatures.ORE_IRON_SMALL), 
            List.of(CountPlacement.of(10), InSquarePlacement.spread(), HeightRangePlacement.uniform(VerticalAnchor.bottom(), VerticalAnchor.absolute(72)), BiomeFilter.biome())));

        // Đồng, Vàng, Đá đỏ, Lapis
        context.register(CURSED_ORE_COPPER, new PlacedFeature(cf.getOrThrow(OreFeatures.ORE_COPPER_LARGE), 
            List.of(CountPlacement.of(16), InSquarePlacement.spread(), HeightRangePlacement.triangle(VerticalAnchor.absolute(-16), VerticalAnchor.absolute(112)), BiomeFilter.biome())));
        context.register(CURSED_ORE_GOLD, new PlacedFeature(cf.getOrThrow(OreFeatures.ORE_GOLD), 
            List.of(CountPlacement.of(4), InSquarePlacement.spread(), HeightRangePlacement.triangle(VerticalAnchor.absolute(-64), VerticalAnchor.absolute(32)), BiomeFilter.biome())));
        context.register(CURSED_ORE_REDSTONE, new PlacedFeature(cf.getOrThrow(OreFeatures.ORE_REDSTONE), 
            List.of(CountPlacement.of(8), InSquarePlacement.spread(), HeightRangePlacement.triangle(VerticalAnchor.absolute(-32), VerticalAnchor.absolute(32)), BiomeFilter.biome())));
        context.register(CURSED_ORE_LAPIS, new PlacedFeature(cf.getOrThrow(OreFeatures.ORE_LAPIS), 
            List.of(CountPlacement.of(2), InSquarePlacement.spread(), HeightRangePlacement.triangle(VerticalAnchor.absolute(-32), VerticalAnchor.absolute(32)), BiomeFilter.biome())));

        // Kim cương (Diamond) - Đầy đủ 4 biến thể
        context.register(CURSED_ORE_DIAMOND_SMALL, new PlacedFeature(cf.getOrThrow(OreFeatures.ORE_DIAMOND_SMALL), 
            List.of(CountPlacement.of(7), InSquarePlacement.spread(), HeightRangePlacement.triangle(VerticalAnchor.aboveBottom(-80), VerticalAnchor.aboveBottom(80)), BiomeFilter.biome())));
        context.register(CURSED_ORE_DIAMOND_MEDIUM, new PlacedFeature(cf.getOrThrow(OreFeatures.ORE_DIAMOND_MEDIUM), 
            List.of(CountPlacement.of(2), InSquarePlacement.spread(), HeightRangePlacement.uniform(VerticalAnchor.absolute(-64), VerticalAnchor.absolute(-4)), BiomeFilter.biome())));
        context.register(CURSED_ORE_DIAMOND_LARGE, new PlacedFeature(cf.getOrThrow(OreFeatures.ORE_DIAMOND_LARGE), 
            List.of(RarityFilter.onAverageOnceEvery(9), InSquarePlacement.spread(), HeightRangePlacement.triangle(VerticalAnchor.aboveBottom(-80), VerticalAnchor.aboveBottom(80)), BiomeFilter.biome())));
        context.register(CURSED_ORE_DIAMOND_BURIED, new PlacedFeature(cf.getOrThrow(OreFeatures.ORE_DIAMOND_BURIED), 
            List.of(CountPlacement.of(4), InSquarePlacement.spread(), HeightRangePlacement.triangle(VerticalAnchor.aboveBottom(-80), VerticalAnchor.aboveBottom(80)), BiomeFilter.biome())));

        // ==========================================
        // C. THẢM THỰC VẬT (Đã sửa lỗi không mọc)
        // ==========================================
        
        // Cây Savanna (Cần HEIGHTMAP và BlockPredicate để mọc trên đất/cỏ)
        context.register(CURSED_TREES_SAVANNA, new PlacedFeature(
                cf.getOrThrow(VegetationFeatures.TREES_SAVANNA),
                List.of(
                        RarityFilter.onAverageOnceEvery(4),
                        CountPlacement.of(1),
                        InSquarePlacement.spread(),
                        PlacementUtils.HEIGHTMAP,
                        BiomeFilter.biome()
                )
        ));

        // Cỏ dại (Cần HEIGHTMAP)
        context.register(CURSED_PATCH_TALL_GRASS, new PlacedFeature(
                cf.getOrThrow(VegetationFeatures.TALL_GRASS),
                List.of(
                        CountPlacement.of(10), 
                        InSquarePlacement.spread(),
                        PlacementUtils.HEIGHTMAP, // 👈 Tìm bề mặt
                        BiomeFilter.biome()
                )
        ));

        // Bụi cây chết (Cần HEIGHTMAP)
        context.register(CURSED_PATCH_DEAD_BUSH, new PlacedFeature(
                cf.getOrThrow(VegetationFeatures.DEAD_BUSH),
                List.of(
                        CountPlacement.of(4), 
                        InSquarePlacement.spread(),
                        PlacementUtils.HEIGHTMAP,
                        BiomeFilter.biome()
                )
        ));
    }

    private static void registerOre(BootstrapContext<PlacedFeature> context, HolderGetter<ConfiguredFeature<?, ?>> cf,
                                    ResourceKey<PlacedFeature> placedKey, ResourceKey<ConfiguredFeature<?, ?>> configuredKey,
                                    int count, int minY, int maxY) {
        var height = TrapezoidHeight.of(VerticalAnchor.absolute(minY), VerticalAnchor.absolute(maxY));
        var modifiers = List.of(
                CountPlacement.of(count),
                InSquarePlacement.spread(),
                HeightRangePlacement.of(height),
                BiomeFilter.biome()
        );
        context.register(placedKey, new PlacedFeature(cf.getOrThrow(configuredKey), modifiers));
    }
}
